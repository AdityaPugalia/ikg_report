  model/.style={stage,draw=cModel,fill=cModel!8},

  kbnote/.style={draw=cConnector,thick,rounded corners=2pt,fill=cConnector!10,align=left,inner sep=3pt,text width=38mm,font=\scriptsize},

  back/.style={->,thick,cModel,dashed}
]
% ---------- left column: steps 1-5 ----------
\node[model] (und) at (0,0) {\textbf{1 Understand} \textcolor{cModel}{[model]}\\ targets + coordinates, formula over targets, givens, unit, analysis statement};
\node[model,below=of und] (map) {\textbf{2 Map words $\to$ entity} \textcolor{cModel}{[model]}\\ candidates by name, filer captions and meaning; a number or \emph{none}; never cached};
\node[model,below=of map] (docs) {\textbf{3 Choose documents} \textcolor{cModel}{[model, or code when one fits]}\\ tools: search documents, company filers, list};
\node[code,below=of docs] (dec) {\textbf{4 Search or derive?} [code]\\ calibrated chance from the bridge, nearest industry first: high $\to$ search; measured 0 $\to$ formulas; unknown $\to$ search, then derive};
\node[model,below=of dec] (routes) {\textbf{5 Routes} \textcolor{cModel}{[model when several]}\\ formulas solved for the target, ranked by trust then joint chance of inputs; ancestors excluded};
% ---------- right column: steps 6-8 ----------
\node[model] (find) at (72mm,0) {\textbf{6 Find} \textcolor{cModel}{[model, rounds $\le 3$]}\\ scope: the parts the bridge names; passages from the retriever; reply: found / found text / collect / widen / search / absent};
\node[code,below=of find] (ground) {\textbf{Ground} [code]\\ quote printed in the passage and row; value and sign from the cell; unit from quoted evidence; kind of number vs kind of quantity};
\node[code,below=of ground] (plan) {\textbf{7 Plan state} [code]\\ settle; propagate up complete routes; switch to a cheaper paused route; abandon inputs no longer needed; evidence kept in full};
\node[model,below=of plan] (resp) {\textbf{8 Respond} \textcolor{cModel}{[model; none for a bare number]}\\ derived values recomputed from cited inputs via the graph's equation; claims must quote evidence; may ask for more targets};
\node[code,below=of resp] (eval) {\textbf{Evaluate} [code; model only to restate]\\ shape, unit, sign fixed; percentage points for rate differences};
% ---------- forward arrows ----------
\draw[arr] (und) -- (map); \draw[arr] (map) -- (docs); \draw[arr] (docs) -- (dec); \draw[arr] (dec) -- (routes);
\draw[arr] (find) -- (ground); \draw[arr] (ground) -- (plan); \draw[arr] (plan) -- (resp); \draw[arr] (resp) -- (eval);
\draw[arr] (dec.east) -- ++(5mm,0) |- node[pos=.75,above,font=\tiny]{search first} ([yshift=2mm]find.west);
\draw[arr] (routes.east) -- ++(9mm,0) |- ([yshift=-2mm]find.west);
% ---------- backtracking ----------
\draw[back] (ground.west) -- ++(-3mm,0) |- node[pos=.25,left,font=\tiny,align=right]{rejected:\\ re-ask once} ([yshift=-6mm]find.west);
\draw[back] (plan.south west) -- ++(-13mm,0) |- node[pos=.3,left,font=\tiny,align=right]{input absent:\\ expand or\\ switch route} ([yshift=-3mm]routes.east);

\node[kbnote,left=6mm of dec] (n3) {\textcolor{cConnector}{\textbf{bridge}}: share of filings of this kind and industry printing the entity, and where};
\node[kbnote,left=6mm of routes] (n4) {\textcolor{cEntity}{\textbf{concept graph}}: equations and subtotals that produce the entity; trust identity $<$ law $<$ empirical};
\node[kbnote,right=6mm of find] (n5) {\textcolor{cConnector}{\textbf{bridge}} $\to$ parts to search first; \textcolor{cDoc}{\textbf{source graph}} $\to$ headings linked to parts; learned retriever ranks chunks};
\node[kbnote,right=6mm of resp] (n6) {\textcolor{cEntity}{\textbf{concept graph}}: the equation reloaded by id for recomputation in code};
